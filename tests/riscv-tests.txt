// Prerequisite : download riscv64-unknown-elf
// Use riscv64-unknown-elf to compile riscv-tests
export RISCV=/path/to/riscv64-unknown-elf
git clone https://github.com/riscv/riscv-tests
cd riscv-tests
git submodule update --init --recursive
autoconf
./configure --prefix=$RISCV/target
make XLEN=32
// Use "python3 tests/run_riscv_tests.py" to run riscv-tests

// it will run tests in riscv-tests/isa for riscv32imacsu 
// Running a single test 
./emu --mode=unit-test --elf=path/to/riscv-tests/isa/rv32ui-p-add
