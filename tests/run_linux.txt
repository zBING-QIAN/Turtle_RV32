# Prerequisite : opensbi, busybox, linux
# opensbi set address of linux and dtb of fw_jump 
git clone git clone https://github.com/riscv-software-src/opensbi.git
cd opensbi
make     CROSS_COMPILE=riscv64-linux-gnu-     PLATFORM=generic     PLATFORM_RISCV_XLEN=32     PLATFORM_RISCV_ISA=rv32ima_zicsr_zifencei     PLATFORM_RISCV_ABI=ilp32     FW_JUMP_ADDR=0x80400000 FW_JUMP_FDT_ADDR=0x80100000

# busybox, compile it with riscv32-unknown-linux-gnu (not riscv32-unknown-elf)
git clone https://github.com/mirror/busybox.git
cd busybox
make CROSS_COMPILE=riscv32-unknown-linux-gnu- ARCH=riscv defconfig
make CROSS_COMPILE=riscv32-unknown-linux-gnu- ARCH=riscv menuconfig
#uncheck network and mail
#[ ] SHA1: Use hardware accelerated instructions if possible              
#[ ] SHA256: Use hardware accelerated instructions if possible 
both under Settings menu
make CROSS_COMPILE=riscv32-unknown-linux-gnu- ARCH=riscv menuconfig
make CROSS_COMPILE=riscv32-unknown-linux-gnu- ARCH=riscv -j$(nproc)
make CROSS_COMPILE=riscv32-unknown-linux-gnu- ARCH=riscv install
cd busybox/_install
mkdir -p sys
mkdir -p proc
mkdir -p dev
sudo mknod -m 600 dev/console c 5 1
sudo mknod -m 666 dev/ttyS0 c 4 64

# put tests/linux-test/init into busybox/_install/init

# rootfs.cpio for linux compile 
cd busybox/_install
find . | cpio -H newc -o > ../rootfs.cpio


####################################################
# Compile linux and pack rootfs.cpio into it
# clone linux version v6.19-rc2-2-gb927546677c8
git clone https://github.com/torvalds/linux.git
cd linux
git checkout v6.19-rc2-2-gb927546677c8
# move tests/linux-test/.config to linux/.config

make CROSS_COMPILE=riscv32-unknown-linux-gnu- ARCH=riscv menuconfig
# Change General setup -> 
# [*] Initial RAM filesystem and RAM disk (initramfs/initrd) support
# (PATH_TO_BUSYBOX/busybox/rootfs.cpio) Initramfs source file(s) 
make CROSS_COMPILE=riscv32-unknown-linux-gnu- ARCH=riscv -j$(nproc)


############################ Prepare DTB ############################
dtc -I dts -O dtb -o tests/linux-test/minimal.dtb tests/linux-test/minimal_rv32.dts


########################## Run Linux ##########################
./emu --mode=linux --linux=/home/hello/workspace/Turtle_RV32/linux/arch/riscv/boot/Image \
--sbi=/home/hello/workspace/Turtle_RV32/opensbi/build/platform/generic/firmware/fw_jump.bin \
--dtb=/home/hello/workspace/Turtle_RV32/tests/linux-test/minimal.dtb
